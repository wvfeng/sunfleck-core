cmake_minimum_required(VERSION 3.24)
project(SunfleckSkiaD3D12Evaluation LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows MSVC is required")
endif()
if(CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰|注意")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(SKIA_ROOT "" CACHE PATH "Fixed Skia checkout")
set(ANGLE_DIRECTORY "" CACHE PATH "Previously evaluated ANGLE directory")
option(SKIA_EVALUATE_ANGLE "Build the optional ANGLE comparison" OFF)
option(SKIA_ALLOW_UNPATCHED_BASELINE "Explicitly permit the original failing candidate for comparison" OFF)
if(NOT SKIA_ALLOW_UNPATCHED_BASELINE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../prepare-d3d12.py"
    --skia "${SKIA_ROOT}" RESULT_VARIABLE candidate_status OUTPUT_VARIABLE candidate_identity ERROR_VARIABLE candidate_error)
  if(NOT candidate_status EQUAL 0)
    message(FATAL_ERROR "D3D12 source candidate rejected: ${candidate_error}")
  endif()
  file(WRITE "${CMAKE_BINARY_DIR}/source-candidate.json" "${candidate_identity}")
endif()

set(D3D12_LIBS "${SKIA_ROOT}/out/sunfleck-d3d12")
set(ANGLE_LIBS "${SKIA_ROOT}/out/sunfleck-gpu-text")
add_executable(d3d12_capabilities ../d3d12-capabilities.cpp)
target_link_libraries(d3d12_capabilities PRIVATE d3d12 dxgi)
add_executable(d3d12_probe ../d3d12.cpp ../text.cpp)
target_include_directories(d3d12_probe PRIVATE "${SKIA_ROOT}")
target_compile_options(d3d12_probe PRIVATE /utf-8)
target_compile_definitions(d3d12_probe PRIVATE SK_GANESH SK_DIRECT3D SKIA_EVALUATION_DWRITE_ONLY)
target_link_libraries(d3d12_probe PRIVATE "${D3D12_LIBS}/skia.lib" "${D3D12_LIBS}/skcms.lib"
  "${D3D12_LIBS}/d3d12allocator.lib" "${D3D12_LIBS}/spirv_cross.lib"
  d3d12 dxgi d3dcompiler dwrite usp10 ole32 user32 gdi32 advapi32 fontsub)
# Equal-scope comparison: both binaries use the same DirectWrite-only source,
# CRT and build type, with no SkParagraph/ICU payload.
if(SKIA_EVALUATE_ANGLE)
add_executable(angle_comparison ../gpu.cpp ../text.cpp)
target_include_directories(angle_comparison PRIVATE "${SKIA_ROOT}"
  "${SKIA_ROOT}/third_party/externals/egl-registry/api"
  "${SKIA_ROOT}/third_party/externals/opengl-registry/api")
target_compile_options(angle_comparison PRIVATE /utf-8)
target_compile_definitions(angle_comparison PRIVATE SK_GANESH SK_GL SKIA_EVALUATION_DWRITE_ONLY)
target_link_libraries(angle_comparison PRIVATE "${ANGLE_LIBS}/skia.lib" "${ANGLE_LIBS}/skcms.lib"
  dwrite usp10 ole32 user32 gdi32 advapi32 fontsub opengl32)
endif()
enable_testing()
add_test(NAME d3d12_warp_semantics COMMAND d3d12_probe --warp "${CMAKE_BINARY_DIR}/warp" --require-debug-layer)
add_test(NAME d3d12_hardware_semantics COMMAND d3d12_probe --hardware "${CMAKE_BINARY_DIR}/hardware" --require-debug-layer)
if(SKIA_EVALUATE_ANGLE)
  add_test(NAME angle_comparison_semantics COMMAND angle_comparison "${ANGLE_DIRECTORY}" "${CMAKE_BINARY_DIR}/angle")
  set_tests_properties(angle_comparison_semantics PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
endif()
set_tests_properties(d3d12_warp_semantics d3d12_hardware_semantics
  PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
install(TARGETS d3d12_probe RUNTIME DESTINATION bin)
install(FILES "${SKIA_ROOT}/LICENSE" DESTINATION share/licenses/skia)
install(FILES "${SKIA_ROOT}/modules/skcms/LICENSE" DESTINATION share/licenses/skcms)
install(FILES "${SKIA_ROOT}/third_party/externals/d3d12allocator/LICENSE.txt" DESTINATION share/licenses/d3d12allocator)
install(FILES "${SKIA_ROOT}/third_party/externals/spirv-cross/LICENSE" DESTINATION share/licenses/spirv-cross)
install(DIRECTORY "${SKIA_ROOT}/third_party/externals/spirv-cross/LICENSES/" DESTINATION share/licenses/spirv-cross/LICENSES)
