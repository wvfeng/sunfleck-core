cmake_minimum_required(VERSION 3.24)
project(SunfleckSkiaWebEvaluation LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Use the existing Emscripten toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(SKIA_ROOT "" CACHE PATH "Pinned Skia checkout")
set(SUNFLECK_ROOT "" CACHE PATH "Sunfleck working tree")
set(SUNFLECK_LIBRARY "" CACHE FILEPATH "Existing Wasm core archive")
set(EVALUATION_FONT "" CACHE FILEPATH "Locally licensed font for private evaluation")
set(SKIA_WEB_LIBS "${SKIA_ROOT}/out/sunfleck-web")
add_library(skia_bridge STATIC bridge.cpp)
target_include_directories(skia_bridge PRIVATE "${SKIA_ROOT}" "${SKIA_ROOT}/third_party/externals/freetype/include")
target_compile_definitions(skia_bridge PRIVATE SK_GANESH SK_GL SK_FORCE_8_BYTE_ALIGNMENT SK_ASSUME_WEBGL=1 SKVX_DISABLE_SIMD)
target_compile_options(skia_bridge PRIVATE -fwasm-exceptions)
target_link_libraries(skia_bridge PUBLIC "${SKIA_WEB_LIBS}/libskia.a" "${SKIA_WEB_LIBS}/libskcms.a"
  "${SKIA_WEB_LIBS}/libfreetype2.a" "${SKIA_WEB_LIBS}/libpng.a" "${SKIA_WEB_LIBS}/libzlib.a")
add_executable(skia_web_app "${SUNFLECK_ROOT}/web/main.cpp")
target_include_directories(skia_web_app PRIVATE "${SUNFLECK_ROOT}/include")
target_compile_options(skia_web_app PRIVATE -fwasm-exceptions)
target_link_libraries(skia_web_app PRIVATE "${SUNFLECK_LIBRARY}" skia_bridge)
target_link_options(skia_web_app PRIVATE -fwasm-exceptions -sUSE_WEBGL2=1 -Wl,--threads=1
  -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web -sALLOW_MEMORY_GROWTH=1
  "-sEXPORTED_FUNCTIONS=['_skia_web_init','_skia_web_present','_skia_web_lost','_skia_web_restore','_skia_web_close','_skia_web_font_version','_skia_web_features']"
  "-sEXPORTED_RUNTIME_METHODS=['HEAPU8','FS','UTF8ToString']"
  "--preload-file=${EVALUATION_FONT}@/sunfleck/font.ttf")
set_target_properties(skia_web_app PROPERTIES OUTPUT_NAME sunfleck SUFFIX .js RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/web")
foreach(asset index.html host.js text-input.js workspace.fleck message.txt)
  configure_file("${SUNFLECK_ROOT}/web/${asset}" "${CMAKE_BINARY_DIR}/web/${asset}" COPYONLY)
endforeach()
configure_file(renderer.js "${CMAKE_BINARY_DIR}/web/renderer.js" COPYONLY)
configure_file("${SUNFLECK_ROOT}/web/renderer.js" "${CMAKE_BINARY_DIR}/web/reference-renderer.js" COPYONLY)
file(READ "${SUNFLECK_ROOT}/web/main.js" main)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cpp-bridge.js" bridge)
string(REPLACE "  await loadHostedFont" "${bridge}\n  await loadHostedFont" main "${main}")
file(WRITE "${CMAKE_BINARY_DIR}/web/main.js" "${main}")
