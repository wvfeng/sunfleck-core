cmake_minimum_required(VERSION 3.24)
project(SkiaEvaluationConsumer LANGUAGES C CXX)
if(CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰|注意")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_path(EVALUATION_INCLUDE evaluation.h REQUIRED)
find_library(EVALUATION_LIBRARY skia_evaluation REQUIRED)
foreach(language c cpp)
    add_executable(consumer_${language} main.${language})
    target_include_directories(consumer_${language} PRIVATE "${EVALUATION_INCLUDE}")
    target_link_libraries(consumer_${language} PRIVATE "${EVALUATION_LIBRARY}")
endforeach()
enable_testing()
add_test(NAME evaluation_installed_c COMMAND consumer_c "${CMAKE_CURRENT_BINARY_DIR}/c-output")
add_test(NAME evaluation_installed_cpp COMMAND consumer_cpp "${CMAKE_CURRENT_BINARY_DIR}/cpp-output")
