cmake_minimum_required(VERSION 3.24)
project(SunfleckSkiaProbe LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "These probes validate the Windows MSVC candidate")
endif()
# CMake 3.29 can decode this toolchain's UTF-8 Chinese /showIncludes prefix
# as GBK. Correct only that recognized prefix; other locales retain detection.
if(CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰|注意")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(SKIA_ROOT "" CACHE PATH "Pinned Skia source checkout")
set(SKIA_LIBRARY "${SKIA_ROOT}/out/sunfleck-cpu/skia.lib" CACHE FILEPATH "CPU Skia library")
get_filename_component(SKIA_LIBRARY_DIR "${SKIA_LIBRARY}" DIRECTORY)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 20)
option(SKIA_EVALUATE_GPU "Build ANGLE/D3D11 evaluation" OFF)
option(SKIA_EVALUATE_TEXT "Build SkParagraph/DirectWrite evaluation" OFF)
set(ANGLE_DIRECTORY "" CACHE PATH "External ANGLE runtime directory (not redistributed)")
add_executable(skia_probe probe.cpp)
target_compile_features(skia_probe PRIVATE cxx_std_20)
target_include_directories(skia_probe PRIVATE "${SKIA_ROOT}")
set_property(TARGET skia_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(skia_probe PRIVATE "${SKIA_LIBRARY}" "${SKIA_LIBRARY_DIR}/skcms.lib" dwrite usp10 ole32 user32 gdi32 advapi32 fontsub)
enable_testing()
add_test(NAME skia_cpu_semantics COMMAND skia_probe "${CMAKE_CURRENT_BINARY_DIR}/probe.ppm")

if(SKIA_EVALUATE_TEXT)
    get_filename_component(SKIA_LIBRARY_DIR "${SKIA_LIBRARY}" DIRECTORY)
    add_library(evaluation_text STATIC text.cpp)
    target_include_directories(evaluation_text PUBLIC "${SKIA_ROOT}")
    target_compile_options(evaluation_text PRIVATE /utf-8)
    target_compile_definitions(evaluation_text PUBLIC SK_GANESH SK_GL SK_UNICODE_ICU_IMPLEMENTATION)
    target_link_libraries(evaluation_text PUBLIC "${SKIA_LIBRARY}" dwrite usp10 ole32 user32 gdi32 advapi32 fontsub opengl32)
    foreach(library skparagraph skshaper skunicode_core skunicode_icu skcms harfbuzz icu)
        target_link_libraries(evaluation_text PUBLIC "${SKIA_LIBRARY_DIR}/${library}.lib")
    endforeach()
    add_executable(skia_text_probe text_main.cpp)
    target_link_libraries(skia_text_probe PRIVATE evaluation_text)
    configure_file("${SKIA_ROOT}/third_party/externals/icu/common/icudtl.dat" "${CMAKE_CURRENT_BINARY_DIR}/icudtl.dat" COPYONLY)
    add_test(NAME skia_text_semantics COMMAND skia_text_probe "${CMAKE_CURRENT_BINARY_DIR}/text")
    add_library(skia_evaluation SHARED evaluation.cpp)
    target_compile_definitions(skia_evaluation PRIVATE SKIA_EVALUATION_EXPORTS)
    target_link_libraries(skia_evaluation PRIVATE evaluation_text)
    install(TARGETS skia_evaluation RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
    install(FILES evaluation.h DESTINATION include)
    install(FILES "${SKIA_ROOT}/third_party/externals/icu/common/icudtl.dat" DESTINATION bin)
    install(FILES "${SKIA_ROOT}/LICENSE" DESTINATION share/licenses/skia)
    install(FILES "${SKIA_ROOT}/modules/skcms/LICENSE" DESTINATION share/licenses/skcms)
    install(FILES "${SKIA_ROOT}/third_party/externals/harfbuzz/COPYING" DESTINATION share/licenses/harfbuzz)
    install(FILES "${SKIA_ROOT}/third_party/externals/harfbuzz/src/ms-use/COPYING" DESTINATION share/licenses/harfbuzz-ms-use)
    install(FILES "${SKIA_ROOT}/third_party/externals/icu/LICENSE" DESTINATION share/licenses/icu)
endif()
if(SKIA_EVALUATE_GPU)
    if(NOT SKIA_EVALUATE_TEXT OR NOT EXISTS "${ANGLE_DIRECTORY}/libEGL.dll")
        message(FATAL_ERROR "GPU evaluation requires text evaluation and an explicit ANGLE_DIRECTORY")
    endif()
    add_executable(skia_gpu_probe gpu.cpp)
    target_include_directories(skia_gpu_probe PRIVATE
        "${SKIA_ROOT}/third_party/externals/egl-registry/api"
        "${SKIA_ROOT}/third_party/externals/opengl-registry/api")
    target_link_libraries(skia_gpu_probe PRIVATE evaluation_text)
    add_test(NAME skia_gpu_semantics COMMAND skia_gpu_probe "${ANGLE_DIRECTORY}" "${CMAKE_CURRENT_BINARY_DIR}/gpu")
    set_tests_properties(skia_gpu_semantics PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
endif()
