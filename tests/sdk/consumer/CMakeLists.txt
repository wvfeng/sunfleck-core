cmake_minimum_required(VERSION 3.25)
project(SunfleckConsumer LANGUAGES C CXX)
find_package(Sunfleck 1.0.0 CONFIG REQUIRED)
include(CTest)
add_executable(consumer_c main.c)
target_link_libraries(consumer_c PRIVATE Sunfleck::sunfleck)
add_test(NAME installed_c COMMAND consumer_c)
if(EMSCRIPTEN)
  return()
endif()
sunfleck_compile_page(consumer_bindings INPUT counter.fleck OUTPUT counter.hpp FORMAT cpp SYMBOL consumer_page)
add_executable(consumer_cpp main.cpp)
add_dependencies(consumer_cpp consumer_bindings)
target_include_directories(consumer_cpp PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(consumer_cpp PRIVATE Sunfleck::sunfleck)
add_test(NAME installed_cpp COMMAND consumer_cpp "${CMAKE_CURRENT_SOURCE_DIR}/counter.fleck")
