    # Export the C ABI only. Keep C++ implementation symbols private.
    set(api_header "${CMAKE_CURRENT_SOURCE_DIR}/include/sunfleck/ui.h")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${api_header}")
    file(STRINGS "${api_header}" declarations REGEX "^[A-Za-z_][A-Za-z_0-9 *]+ ui_[a-z_]+\\(")
    set(exports "EXPORTS\n")
    foreach(declaration IN LISTS declarations)
      string(REGEX MATCH "ui_[a-z_]+\\(" symbol "${declaration}")
      string(REPLACE "(" "" symbol "${symbol}")
      string(APPEND exports "  ${symbol}\n")
    endforeach()
    set(exports_file "${CMAKE_CURRENT_BINARY_DIR}/sunfleck_native.def")
    file(CONFIGURE OUTPUT "${exports_file}" CONTENT "${exports}" @ONLY)
    target_sources(sunfleck_native PRIVATE "${exports_file}")
  endif()
